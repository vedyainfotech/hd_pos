class MenuSchedule(Base):
    __tablename__ = "menu_schedules"

    __table_args__ = (
        UniqueConstraint(
            "delivery_date",
            "category_id",
            name="uq_menu_schedule_date_category",

        BigInteger,
        primary_key=True,

    )

        Date,
        nullable=False,
    )

    category_id: Mapped[int] = mapped_column(
        BigInteger,
        ForeignKey(
            "categories.id",
            onupdate="CASCADE",
            ondelete="RESTRICT",
        ),
        nullable=False,
    )

    created_at: Mapped[datetime] = mapped_column(
        DateTime,
        nullable=False,
        server_default=func.now(),
    )

    updated_at: Mapped[datetime] = mapped_column(
        DateTime,
        nullable=False,
        server_default=func.now(),
        onupdate=func.now(),
    )

from sqlalchemy.orm import Session
from app.models.hd_items import HDItem


def get_items(
    db: Session,
    include_inactive: bool = False,
    category_id: int | None = None,
    sub_category_id: int | None = None,
    search: str | None = None,
):
    query = db.query(HDItem)

    if not include_inactive:
        query = query.filter(HDItem.status.is_(True))

    if category_id is not None:
        query = query.filter(HDItem.category_id == category_id)

    if sub_category_id is not None:
        query = query.filter(HDItem.sub_category_id == sub_category_id)

    if search:
        query = query.filter(HDItem.name.ilike(f"%{search.strip()}%"))

    return query.order_by(HDItem.id.desc()).all()


def get_item(db: Session, item_id: int):
    return db.query(HDItem).filter(HDItem.id == item_id).first()


def get_item_by_name(db: Session, name: str):
    return (
        db.query(HDItem)
        .filter(HDItem.name.ilike(name.strip()))
        .first()
    )


def create_item(db: Session, item_data: dict):
    item = HDItem(**item_data)
    db.add(item)
    db.commit()
    db.refresh(item)
    return item


def update_item(db: Session, item: HDItem, item_data: dict):
    for key, value in item_data.items():
        setattr(item, key, value)

    db.commit()
    db.refresh(item)
    return item


def update_item_status(db: Session, item: HDItem, status: bool):
    item.status = status
    db.commit()
    db.refresh(item)
    return item

from datetime import date

from sqlalchemy import delete, select
from sqlalchemy.orm import Session

from app.models.menu_schedule import MenuSchedule, MenuScheduleItem
from app.models.category import Category
from app.models.sub_category import SubCategory
from app.models.hd_items import HDItem


# ---------------------------------
# Get Schedule by Date and Category
# ---------------------------------

def get_schedule_by_date_category(
    db: Session,
    delivery_date: date,
    category_id: int,
):
    statement = select(MenuSchedule).where(
        MenuSchedule.delivery_date == delivery_date,
        MenuSchedule.category_id == category_id,
    )

    return db.scalar(statement)


# ---------------------------------
# Get All Schedules for a Date
# ---------------------------------

def get_schedules_by_date(
    db: Session,
    delivery_date: date,
):
    statement = (
        select(MenuSchedule)
        .where(MenuSchedule.delivery_date == delivery_date)
        .order_by(MenuSchedule.category_id)
    )

    return db.scalars(statement).all()


# ---------------------------------
# Get Schedule Items
# ---------------------------------

def get_schedule_items(
    db: Session,
    schedule_id: int,
):
    statement = (
        select(
            MenuScheduleItem.id.label("schedule_item_id"),
            HDItem.id.label("item_id"),
            HDItem.name.label("item_name"),
            HDItem.price.label("price"),
            HDItem.sub_category_id.label("sub_category_id"),
            SubCategory.name.label("sub_category_name"),
        )
        .join(
            HDItem,
            MenuScheduleItem.item_id == HDItem.id,
        )
        .outerjoin(
            SubCategory,
            HDItem.sub_category_id == SubCategory.id,
        )
        .where(
            MenuScheduleItem.schedule_id == schedule_id
        )
        .order_by(
            HDItem.sub_category_id,
            HDItem.name,
        )
    )

    return db.execute(statement).all()

def get_active_category(

    category_id: int,
):
    statement = select(Category).where(
        Category.id == category_id,
        Category.status.is_(True),
    )

    return db.scalar(statement)

def get_available_items(

    category_id: int,
    sub_category_id: int | None = None,

        HDItem.category_id == category_id,
        HDItem.status.is_(True),

    if sub_category_id is not None:

            HDItem.sub_category_id == sub_category_id
        )

    if search:
        statement = statement.where(
            HDItem.name.ilike(f"%{search.strip()}%")
        )

    statement = statement.order_by(HDItem.name)

    return db.scalars(statement).all()

    db: Session,
    item_ids: list[int],
):
    if not item_ids:
        return []

    statement = select(HDItem).where(
        HDItem.id.in_(item_ids)
    )

    return db.scalars(statement).all()

    category_id: int,

# ---------------------------------
# Delete Existing Schedule Items
# ---------------------------------

def delete_schedule_items(
    db: Session,
    schedule_id: int,
):
    # Use a direct SQL DELETE instead of marking ORM objects as deleted.
    # This ensures rows are removed from PostgreSQL before replacement
    # MenuScheduleItem rows with the same (schedule_id, item_id) are inserted.
    statement = delete(MenuScheduleItem).where(
        MenuScheduleItem.schedule_id == schedule_id
    )

    db.execute(
        statement.execution_options(synchronize_session=False)
    )


# ---------------------------------
# Add Schedule Items
# ---------------------------------

def add_schedule_items(
    db: Session,
    schedule_id: int,
    item_ids: list[int],
):
    schedule_items = [
        MenuScheduleItem(
            schedule_id=schedule_id,
            item_id=item_id,
        )
        for item_id in item_ids
    ]

    db.add_all(schedule_items)


# ---------------------------------
# Delete Empty Schedule
# ---------------------------------

def delete_schedule(
    db: Session,
    schedule: MenuSchedule,
):
    db.delete(schedule)

from fastapi import HTTPException
from sqlalchemy.orm import Session

from app.repositories import hd_items as repository


def list_items(
    db: Session,
    include_inactive: bool = False,
    category_id: int | None = None,
    sub_category_id: int | None = None,
    search: str | None = None,
):
    return repository.get_items(
        db=db,
        include_inactive=include_inactive,
        category_id=category_id,
        sub_category_id=sub_category_id,
        search=search,
    )


def create_item(db: Session, item_data: dict):
    name = item_data["name"].strip()

    if not name:
        raise HTTPException(status_code=400, detail="Item name is required")

    if repository.get_item_by_name(db, name):
        raise HTTPException(
            status_code=409,
            detail="Item already exists",
        )

    item_data["name"] = name
    return repository.create_item(db, item_data)


def update_item(db: Session, item_id: int, item_data: dict):
    item = repository.get_item(db, item_id)

    if not item:
        raise HTTPException(status_code=404, detail="Item not found")

    name = item_data["name"].strip()

    if not name:
        raise HTTPException(status_code=400, detail="Item name is required")

    duplicate = repository.get_item_by_name(db, name)

    if duplicate and duplicate.id != item_id:
        raise HTTPException(
            status_code=409,
            detail="Item already exists",
        )

    item_data["name"] = name
    return repository.update_item(db, item, item_data)


def update_status(db: Session, item_id: int, status: bool):
    item = repository.get_item(db, item_id)

    if not item:
        raise HTTPException(status_code=404, detail="Item not found")

    return repository.update_item_status(db, item, status)

from datetime import date

from fastapi import HTTPException, status
from sqlalchemy.orm import Session

from app.models.category import Category
from app.repositories import menu_schedule as repository


# ---------------------------------
# Get Available Menu Items
# ---------------------------------

def get_available_menu(
    db: Session,
    category_id: int,
    search: str | None = None,
):
    category = repository.get_active_category(
        db,
        category_id,
    )

    if not category:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Category not found or inactive",
        )

    subcategories = repository.get_active_subcategories(
        db,
        category_id,
    )

    result = []

    for subcategory in subcategories:
        items = repository.get_available_items(
            db=db,
            category_id=category_id,
            sub_category_id=subcategory.id,
            search=search,
        )

        result.append({
            "id": subcategory.id,
            "name": subcategory.name,
            "items": [
                {
                    "id": item.id,
                    "name": item.name,
                    "price": item.price,
                    "category_id": item.category_id,
                    "sub_category_id": item.sub_category_id,
                    "sub_category_name": subcategory.name,
                }
                for item in items
            ],
        })

    return {
        "category_id": category.id,
        "category_name": category.name,
        "sub_categories": result,
    }


# ---------------------------------
# Validate Selected Items
# ---------------------------------

def validate_selected_items(
    db: Session,
    category_id: int,
    item_ids: list[int],
):
    if len(item_ids) != len(set(item_ids)):
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Duplicate items are not allowed",
        )

    if not item_ids:
        return []

    items = repository.get_items_by_ids(
        db,
        item_ids,
    )

    if len(items) != len(item_ids):
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="One or more selected items do not exist",
        )

    subcategories = repository.get_active_subcategories(
        db,
        category_id,
    )

    active_subcategory_ids = {
        subcategory.id
        for subcategory in subcategories
    }

    for item in items:
        if not item.status:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=f"Item '{item.name}' is inactive",
            )

        if item.category_id != category_id:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=(
                    f"Item '{item.name}' does not belong "
                    "to the selected category"
                ),
            )

        if item.sub_category_id not in active_subcategory_ids:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=(
                    f"Item '{item.name}' does not belong "
                    "to an active subcategory"
                ),
            )

    return items


# ---------------------------------
# Save or Update Menu Schedule
# ---------------------------------

def save_menu_schedule(
    db: Session,
    delivery_date: date,
    category_id: int,
    item_ids: list[int],
):
    category = repository.get_active_category(
        db,
        category_id,
    )

    if not category:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Category not found or inactive",
        )

    # Validate all selected items before changing the schedule.
    validate_selected_items(
        db,
        category_id,
        item_ids,
    )

    try:
        schedule = repository.get_schedule_by_date_category(
            db,
            delivery_date,
            category_id,
        )

        # No selected items: remove an existing schedule.
        if not item_ids:
            if schedule:
                repository.delete_schedule_items(
                    db,
                    schedule.id,
                )
                # Ensure child rows are deleted before deleting the schedule.
                db.flush()
                repository.delete_schedule(
                    db,
                    schedule,
                )

            db.commit()

            return {
                "message": "Menu schedule cleared successfully",
                "delivery_date": delivery_date,
                "category_id": category_id,
            }

        # Create a new schedule when one doesn't exist.
        if not schedule:
            schedule = repository.create_schedule(
                db,
                delivery_date,
                category_id,
            )

        # Replace the previous item selection.
        # The repository performs a direct SQL DELETE. Flush it before
        # inserting replacement rows to avoid uq_menu_schedule_item conflicts.
        repository.delete_schedule_items(
            db,
            schedule.id,
        )
        db.flush()

        repository.add_schedule_items(
            db,
            schedule.id,
            item_ids,
        )

        db.commit()
        db.refresh(schedule)

        return {
            "message": "Menu schedule saved successfully",
            "schedule_id": schedule.id,
            "delivery_date": schedule.delivery_date,
            "category_id": schedule.category_id,
        }

    except Exception:
        db.rollback()
        raise


# ---------------------------------
# Get Scheduled Menu for a Date
# ---------------------------------

def get_menu_by_date(
    db: Session,
    delivery_date: date,
):
    schedules = repository.get_schedules_by_date(
        db,
        delivery_date,
    )

    result = []

    for schedule in schedules:
        category = db.get(
            Category,
            schedule.category_id,
        )

        if not category:
            continue

        scheduled_items = repository.get_schedule_items(
            db,
            schedule.id,
        )

        items = []

        for item in scheduled_items:
            items.append({
                "id": item.schedule_item_id,
                "item_id": item.item_id,
                "item_name": item.item_name,
                "price": float(item.price),
                "sub_category_id": item.sub_category_id,
                "sub_category_name": item.sub_category_name,
            })

        result.append({
            "id": schedule.id,
            "delivery_date": schedule.delivery_date,
            "category_id": schedule.category_id,
            "category_name": category.name,
            "items": items,
        })

    return {
        "delivery_date": delivery_date,
        "schedules": result,
    }
